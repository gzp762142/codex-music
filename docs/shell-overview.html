enum AppPhase { case loading, unlock, control }

final class AppState: ObservableObject {
  @Published var phase: AppPhase = .loading
  @Published var isPoweredOn = false   // 默认关，卡密通过后仍须手动开
  @Published var sessionStart: Date?
  private let volumeMonitor = VolumeButtonMonitor()
  // Vol+ → FangUIBridge.setVisible(true)  仅 control 阶段
  // Vol- → FangUIBridge.setVisible(false)
}

RootView.body:
  phase == .loading  → LoadingView { completeLoading() }
  phase == .unlock   → UnlockHostView { enterControl() }
  else               → ControlView().environmentObject(state)
  // 前台通知 → resyncOverlay()：control 只保 monitor，其它清 overlay
